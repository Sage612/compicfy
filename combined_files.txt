// ── Pre-fill URL builder for external results ─────────────────────

function buildPrefillUrl(result: ExternalResult, isPro: boolean): string {
  const base = isPro ? '/pro/recommendations/new' : '/dashboard/recommendations/new'
  const params = new URLSearchParams({
    title:       result.title       ?? '',
    author:      result.author      ?? '',
    artist:      result.artist      ?? '',
    description: result.description?.slice(0, 2000) ?? '',
    type:        result.type        ?? 'manga',
    status:      result.status      ?? 'ongoing',
    cover_url:   result.coverUrl    ?? '',
    ...(result.year     ? { year_released: String(result.year)     } : {}),
    ...(result.chapters ? { chapter_count: String(result.chapters) } : {}),
    genres:      (result.genres ?? []).slice(0, 5).join(','),
  })
  return `${base}?${params.toString()}`
}

// ── Main Component ────────────────────────────────────────────────

interface Props {
  initialQuery: string
  isPro:        boolean
}

export function SearchPageClient({ initialQuery, isPro }: Props) {
  const router = useRouter()

  const [query,           setQuery]           = useState(initialQuery)
  const [inputValue,      setInputValue]      = useState(initialQuery)
  const [dbResults,       setDbResults]       = useState<DBResult[]>([])
  const [externalResults, setExternalResults] = useState<ExternalResult[]>([])
  const [loadingDB,       setLoadingDB]       = useState(false)
  const [loadingExternal, setLoadingExternal] = useState(false)
  const [hasSearched,     setHasSearched]     = useState(!!initialQuery)
  const [suggestions,     setSuggestions]     = useState<DBResult[]>([])
  const [showSuggestions, setShowSuggestions] = useState(false)
  const [activeTab,       setActiveTab]       = useState<'all' | 'compicfy' | 'web'>('all')

  const inputRef        = useRef<HTMLInputElement>(null)
  const debounceRef     = useRef<ReturnType<typeof setTimeout> | null>(null)
  const suggDebounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
  const containerRef    = useRef<HTMLDivElement>(null)

  // Close suggestions on outside click
  useEffect(() => {
    const handler = (e: MouseEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setShowSuggestions(false)
      }
    }
    document.addEventListener('mousedown', handler)
    return () => document.removeEventListener('mousedown', handler)
  }, [])

  // Run search on initial query
  useEffect(() => {
    if (initialQuery.trim().length >= 2) runSearch(initialQuery)
  }, [])

  // Live suggestions (DB only, fast)
  const fetchSuggestions = useCallback(async (q: string) => {
    if (q.trim().length < 2) { setSuggestions([]); setShowSuggestions(false); return }
    try {
      const res  = await fetch(`/api/search?q=${encodeURIComponent(q)}&limit=5`)
      const data = await res.json()
      setSuggestions(data.results ?? [])
      setShowSuggestions(true)
    } catch {
      setSuggestions([])
    }
  }, [])

  // Full search (DB + external)
  const runSearch = useCallback(async (q: string) => {
    const trimmed = q.trim()
    if (trimmed.length < 2) return

    setShowSuggestions(false)
    setHasSearched(true)
    setQuery(trimmed)

    // Update URL without navigation
    const url = new URL(window.location.href)
    url.searchParams.set('q', trimmed)
    window.history.replaceState({}, '', url.toString())

    // DB search — fast
    setLoadingDB(true)
    fetch(`/api/search?q=${encodeURIComponent(trimmed)}&limit=20`)
      .then(r => r.json())
      .then(d => setDbResults(d.results ?? []))
      .catch(() => setDbResults([]))
      .finally(() => setLoadingDB(false))

    // External search — slower
    setLoadingExternal(true)
    fetch(`/api/manga-search?q=${encodeURIComponent(trimmed)}`)
      .then(r => r.json())
      .then(d => setExternalResults(d.results ?? []))
      .catch(() => setExternalResults([]))
      .finally(() => setLoadingExternal(false))
  }, [])

  const handleInputChange = (value: string) => {
    setInputValue(value)
    // Live suggestions
    if (suggDebounceRef.current) clearTimeout(suggDebounceRef.current)
    suggDebounceRef.current = setTimeout(() => fetchSuggestions(value), 200)
  }

  const handleSearch = () => {
    if (inputValue.trim().length < 2) return
    runSearch(inputValue)
  }

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter') handleSearch()
    if (e.key === 'Escape') setShowSuggestions(false)
  }

  const handleSuggestionClick = (result: DBResult) => {
    setShowSuggestions(false)
    router.push(`/comic/${result.id}`)
  }

  const handleClear = () => {
    setInputValue('')
    setQuery('')
    setDbResults([])
    setExternalResults([])
    setHasSearched(false)
    setSuggestions([])
    setShowSuggestions(false)
    inputRef.current?.focus()
  }

  // Deduplicate external results against DB results
  const dbTitles = new Set(dbResults.map(r => r.title.toLowerCase().trim()))
  const filteredExternal = externalResults.filter(
    r => !dbTitles.has(r.title.toLowerCase().trim())
  )

  const totalResults = dbResults.length + filteredExternal.length
  const isLoading    = loadingDB || loadingExternal

  return (
    <div className="min-h-[calc(100vh-56px)] bg-background">

      {/* ── Search hero — only shown before first search ── */}
      {!hasSearched && (
        <div className="flex flex-col items-center justify-center pt-24 pb-12 px-4 text-center space-y-4">
          <div className="space-y-2">
           
            <h1 className="text-4xl sm:text-5xl font-black tracking-tight">
              Find your next obsession...
            </h1>
            <p className="text-muted-foreground text-sm max-w-sm mx-auto">
              Search across Compicfy, MyAnimeList, MangaDex, AniList and MangaUpdates in one go.
            </p>
          </div>
        </div>
      )}

      {/* ── Search bar ── */}
      <div className={cn(
        'sticky top-14 z-30 px-3 sm:px-4 py-3 bg-background/95 backdrop-blur-sm border-b border-border transition-all',
        !hasSearched && 'static border-0 bg-transparent backdrop-blur-none pb-8'
      )}>
        <div className="max-w-2xl mx-auto w-full" ref={containerRef}>
          <div className="relative w-full">
            <div className={cn(
              'flex items-center gap-2 sm:gap-3 rounded-2xl border bg-card transition-all w-full min-w-0',
              hasSearched
                ? 'px-3 sm:px-4 py-2.5 border-border shadow-sm'
                : 'px-4 sm:px-5 py-4 border-primary/20 shadow-xl shadow-primary/10'
            )}>
              <Search
                size={hasSearched ? 15 : 18}
                className="text-muted-foreground shrink-0 transition-all"
              />
              <input
                ref={inputRef}
                value={inputValue}
                onChange={e => handleInputChange(e.target.value)}
                onKeyDown={handleKeyDown}
                onFocus={() => suggestions.length > 0 && setShowSuggestions(true)}
                placeholder="Search manga, manhwa…"
                className={cn(
                  'flex-1 bg-transparent outline-none placeholder:text-muted-foreground text-foreground min-w-0 w-0',
                  hasSearched ? 'text-sm' : 'text-base'
                )}
                autoFocus={!initialQuery}
              />
              {isLoading && (
                <Loader2 size={14} className="animate-spin text-muted-foreground shrink-0" />
              )}
              {inputValue && !isLoading && (
                <button onClick={handleClear} className="text-muted-foreground hover:text-foreground transition-colors p-1 shrink-0">
                  <X size={14} />
                </button>
              )}
              <Button
                onClick={handleSearch}
                disabled={inputValue.trim().length < 2}
                size="sm"
                className="shrink-0 h-8 px-4 bg-gradient-to-r from-primary to-violet-500 hover:opacity-90 border-0 font-bold"
              >
                Search
              </Button>
            </div>

            {/* Live suggestions dropdown */}
            {showSuggestions && suggestions.length > 0 && (
              <div className="absolute top-full left-0 right-0 mt-2 rounded-2xl border border-border bg-card shadow-2xl shadow-black/20 overflow-hidden z-50">
                <div className="py-1.5">
                  {suggestions.map((s, i) => {
                    const typeColor = TYPE_COLORS[s.type] ?? TYPE_COLORS.other
                    return (
                      <button
                        key={s.id}
                        onClick={() => handleSuggestionClick(s)}
                        className="w-full flex items-center gap-3 px-4 py-2.5 hover:bg-muted/50 transition-colors text-left"
                      >
                        <div className="w-7 h-10 rounded-lg overflow-hidden bg-muted shrink-0">
                          {s.cover_url ? (
                            <img src={s.cover_url} alt={s.title} className="w-full h-full object-cover" />
                          ) : (
                            <div className="w-full h-full flex items-center justify-center">
                              <BookOpen size={10} className="text-muted-foreground opacity-40" />
                            </div>
                          )}
                        </div>
                        <div className="flex-1 min-w-0">
                          <p className="text-sm font-semibold truncate">{s.title}</p>
                          <div className="flex items-center gap-1.5 mt-0.5">
                            <span className={cn('text-[9px] font-bold px-1.5 py-0.5 rounded-full capitalize border', typeColor)}>
                              {s.type}
                            </span>
                            {s.author && (
                              <span className="text-[10px] text-muted-foreground truncate">{s.author}</span>
                            )}
                          </div>
                        </div>
                        <ArrowRight size={11} className="text-muted-foreground/30 shrink-0" />
                      </button>
                    )
                  })}
                  {/* Search all option */}
                  <button
                    onClick={handleSearch}
                    className="w-full flex items-center gap-3 px-4 py-2.5 border-t border-border hover:bg-primary/5 transition-colors text-left mt-1"
                  >
                    <div className="w-7 h-10 rounded-lg bg-primary/10 flex items-center justify-center shrink-0">
                      <Search size={10} className="text-primary" />
                    </div>
                    <p className="text-sm font-semibold text-primary">
                      Search all sources for "{inputValue}"
                    </p>
                  </button>
                </div>
              </div>
            )}
          </div>

          {/* Quick links — only before first search */}
          {!hasSearched && (
            <div className="mt-6 space-y-3">
              <p className="text-[10px] font-black uppercase tracking-widest text-muted-foreground text-center">
                Quick browse
              </p>
              <div className="flex flex-wrap gap-2 justify-center">
                {QUICK_LINKS.map(link => (
                  <Link
                    key={link.href}
                    href={link.href}
                    className="px-3 py-1.5 rounded-full border border-border bg-card text-xs font-medium text-muted-foreground hover:text-foreground hover:border-primary/40 hover:bg-primary/5 transition-all"
                  >
                    {link.label}
                  </Link>
                ))}
              </div>
            </div>
          )}
        </div>
      </div>

      {/* ── Results ── */}
      {hasSearched && (
        <div className="max-w-3xl mx-auto px-4 py-6 space-y-6">

          {/* Result count + tabs */}
          {!isLoading && totalResults > 0 && (
            <div className="flex items-center justify-between flex-wrap gap-3">
              <p className="text-sm text-muted-foreground">
                <span className="font-black text-foreground">{totalResults}</span> results for{' '}
                <span className="font-semibold">"{query}"</span>
              </p>
              <div className="flex gap-1">
                {[
                  { key: 'all',      label: 'All',      count: totalResults          },
                  { key: 'compicfy', label: 'Compicfy', count: dbResults.length      },
                  { key: 'web',      label: 'Web',      count: filteredExternal.length },
                ].map(tab => (
                  <button
                    key={tab.key}
                    onClick={() => setActiveTab(tab.key as typeof activeTab)}
                    className={cn(
                      'px-3 py-1.5 rounded-full text-xs font-bold border transition-all',
                      activeTab === tab.key
                        ? 'bg-primary text-primary-foreground border-primary shadow-sm'
                        : 'border-border text-muted-foreground hover:border-primary/50'
                    )}
                  >
                    {tab.label} {tab.count > 0 && `(${tab.count})`}
                  </button>
                ))}
              </div>
            </div>
          )}

          {/* DB results */}
          {(activeTab === 'all' || activeTab === 'compicfy') && (
            <div className="space-y-3">
              {(activeTab === 'all' || activeTab === 'compicfy') && dbResults.length > 0 && (
                <div className="flex items-center gap-2">
                  <div className="w-5 h-5 rounded-md bg-primary/10 flex items-center justify-center">
                    <Database size={10} className="text-primary" />
                  </div>
                  <p className="text-xs font-black uppercase tracking-widest text-muted-foreground">
                    In Compicfy
                  </p>
                  <div className="flex-1 h-px bg-border" />
                </div>
              )}
              {loadingDB ? (
                <div className="space-y-3">
                  {[1,2,3].map(i => (
                    <div key={i} className="h-28 rounded-2xl bg-muted/40 animate-pulse" />
                  ))}
                </div>
              ) : dbResults.length > 0 ? (
                <div className="space-y-2">
                  {dbResults.map(r => <DBCard key={r.id} result={r} />)}
                </div>
              ) : hasSearched && !loadingDB && activeTab === 'compicfy' ? (
                <div className="text-center py-10 rounded-2xl border border-dashed border-border space-y-2">
                  <Database size={24} className="text-muted-foreground/30 mx-auto" />
                  <p className="text-sm font-semibold">Not in Compicfy yet</p>
                  <p className="text-xs text-muted-foreground">Check the Web tab to find it and add it</p>
                </div>
              ) : null}
            </div>
          )}

          {/* External results */}
          {(activeTab === 'all' || activeTab === 'web') && (
            <div className="space-y-3">
              {filteredExternal.length > 0 && (
                <div className="flex items-center gap-2">
                  <div className="w-5 h-5 rounded-md bg-violet-500/10 flex items-center justify-center">
                    <Globe size={10} className="text-violet-500" />
                  </div>
                  <p className="text-xs font-black uppercase tracking-widest text-muted-foreground">
                    From the web
                  </p>
                  <div className="flex-1 h-px bg-border" />
                  {loadingExternal && (
                    <Loader2 size={11} className="animate-spin text-muted-foreground" />
                  )}
                </div>
              )}

              {loadingExternal && filteredExternal.length === 0 ? (
                <div className="space-y-3">
                  {[1,2,3].map(i => (
                    <div key={i} className="h-28 rounded-2xl bg-muted/40 animate-pulse" />
                  ))}
                </div>
              ) : filteredExternal.length > 0 ? (

// Hardcoded fallback in case env var is missing

export async function POST(req: NextRequest) {
  try {
    const supabase = await createClient()
    const { data: { user } } = await supabase.auth.getUser()

    if (!user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    }

    const { plan } = await req.json()

    if (!plan || !['monthly', 'yearly'].includes(plan)) {
      return NextResponse.json({ error: 'Invalid plan' }, { status: 400 })
    }

    const { data: profile } = await (supabase as any)
      .from('profiles')
      .select('username, display_name, is_pro, pro_expires_at, role')
      .eq('id', user.id)
      .single()

    if (!profile) {
      return NextResponse.json({ error: 'Profile not found' }, { status: 404 })
    }

    if (profile.role === 'admin') {
      return NextResponse.json({ error: 'Admins have free Pro access' }, { status: 400 })
    }

    const selectedPlan = PAYSTACK_PLANS[plan as keyof typeof PAYSTACK_PLANS]

    // Always use the hardcoded APP_URL as primary, env var as override
    const callbackUrl = `${APP_URL}/api/paystack/verify`

    console.log('Paystack callback URL:', callbackUrl)
    console.log('Paystack plan:', plan, 'amount:', selectedPlan.amount)

    const data = await paystackRequest('/transaction/initialize', 'POST', {
      email: user.email,
      // amount is already in smallest currency unit (kobo for NGN, cents for USD)
      // DO NOT multiply by 100 again
      amount: selectedPlan.amount,
      currency: 'USD',
      callback_url: callbackUrl,
      metadata: {
        user_id: user.id,
        plan,
        username: profile.username,
        custom_fields: [
          {
            display_name: 'Plan',
            variable_name: 'plan',
            value: selectedPlan.label,
          },
        ],
      },
      channels: ['card'],
    })

    return NextResponse.json({
      authorization_url: data.data.authorization_url,
      access_code: data.data.access_code,
      reference: data.data.reference,
    })
  } catch (error) {
    console.error('Paystack initialize error:', error)
    return NextResponse.json({ error: 'Failed to initialize payment' }, { status: 500 })
  }

import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'
import { paystackRequest } from '@/lib/paystack/client'

interface ProfileResult {
  username: string
  display_name: string | null
  is_pro: boolean
}

const APP_URL = process.env.NEXT_PUBLIC_APP_URL?.replace(/\/$/, '') || 'https://compicfy.cloud'

export async function GET(req: NextRequest) {
  try {
    const { searchParams } = new URL(req.url)
    const reference = searchParams.get('reference')

    if (!reference) {
      return NextResponse.redirect(`${APP_URL}/pricing?error=missing_reference`)
    }

    // Verify transaction with Paystack
    let data: any
    try {
      data = await paystackRequest(`/transaction/verify/${reference}`)
    } catch {
      return NextResponse.redirect(`${APP_URL}/pricing?error=verification_failed`)
    }

    if (data.data.status !== 'success') {
      return NextResponse.redirect(`${APP_URL}/pricing?error=payment_failed`)
    }

    const metadata = data.data.metadata
    const userId   = metadata?.user_id
    const plan     = metadata?.plan

    if (!userId || !plan) {
      return NextResponse.redirect(`${APP_URL}/pricing?error=invalid_metadata`)
    }

    const supabase = await createClient()

    // Guard: check if this reference was already processed to prevent double-activation
    const { data: existingSub } = await supabase
      .from('subscriptions')
      .select('id')
      .eq('paystack_reference', reference)
      .maybeSingle()

    if (!existingSub) {
      // Only activate if not already processed
      await (supabase as any).rpc('activate_pro', {
        p_user_id: userId,
        p_plan: plan,
        p_paystack_reference: reference,
        p_paystack_customer_code: data.data.customer?.customer_code ?? null,
      })
    }

    // Get profile for Discord notification
    const { data: profileData } = await supabase
      .from('profiles')
      .select('username, display_name, is_pro')
      .eq('id', userId)
      .single()

    const profile = profileData as ProfileResult | null

    // Fire Discord webhook
    if (process.env.DISCORD_PRO_WEBHOOK_URL) {
      try {
        const displayName = profile?.display_name || profile?.username || 'A user'
        await fetch(process.env.DISCORD_PRO_WEBHOOK_URL, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            embeds: [{
              title: '👑 New Pro Subscriber!',
              description: `**${displayName}** just upgraded to Compicfy Pro (${plan} plan)! 🎉`,
              color: 0xF59E0B,
              timestamp: new Date().toISOString(),
              footer: { text: 'Compicfy Pro' },
            }],
          }),
        })
      } catch (e) {
        console.error('Discord webhook failed:', e)
      }
    }

    return NextResponse.redirect(`${APP_URL}/pro/dashboard?welcome=true`)
  } catch (error) {
    console.error('Paystack verify error:', error)
    return NextResponse.redirect(`${APP_URL}/pricing?error=verification_failed`)
  }
}

        </div>

        <div className="px-6 pb-6 -mt-12 space-y-5">
          <div className="flex flex-col sm:flex-row sm:items-end justify-between gap-4">
            <div className="flex flex-col sm:flex-row sm:items-end gap-4">
              <ProAvatar
                src={profile.avatar_url}
                fallback={initials}
                size="xl"
                isPro={isPro}
                isAdmin={isAdmin}
                className="border-4 border-card shadow-lg shrink-0"
              />
              <div className="space-y-1">
                <div className="flex flex-wrap items-center gap-2">
                  <h1 className="text-2xl font-black truncate">
                    {profile.display_name || profile.username}
                  </h1>
                  <ProBadge role={profile.role} isPro={isPro} />
                  {profile.verified_recommender && <VerifiedBadge />}
                </div>
                <p className="text-sm text-muted-foreground">@{profile.username}</p>
              </div>
            </div>

            {/* Follow button — hidden for ghost and for self */}
            {!isOwner && !isGhost && viewer && (
              <div className="shrink-0 sm:pb-1">
                <FollowButton
                  profileId={profile.id}
                  initialFollowing={isFollowing}
                  isLoggedIn={!!viewer}
                />
              </div>
            )}

            {/* Ghost — show lore pill instead of follow button */}
            {isGhost && (
              <div className="shrink-0 sm:pb-1">
                <div className="inline-flex items-center gap-2 px-3 py-1.5 rounded-full border border-indigo-500/30 bg-indigo-500/10 text-indigo-400 text-xs font-semibold">
                  <span className="w-1.5 h-1.5 rounded-full bg-indigo-400 animate-pulse" />
                  The Archivist
                </div>
              </div>
            )}
          </div>

          {profile.bio && (
            <p className="text-sm leading-relaxed text-muted-foreground max-w-lg">{profile.bio}</p>
          )}

          <div className="flex flex-wrap gap-4">
            {/* Ghost doesn't show followers/following count */}
            {!isGhost && (
              <>
                <Link
                  href={`/profile/${profile.username}`}
                  className="flex items-center gap-1.5 text-sm text-muted-foreground hover:text-foreground transition-colors"
                >
                  <Users size={13} className="text-primary/60" />
                  <span><strong className="text-foreground font-black">{followersCount ?? 0}</strong> followers</span>
                </Link>
                <div className="flex items-center gap-1.5 text-sm text-muted-foreground">
                  <Users size={13} className="text-primary/60" />
                  <span><strong className="text-foreground font-black">{followingCount ?? 0}</strong> following</span>
                </div>
              </>
            )}
            {[
              { icon: BookOpen, value: profile.total_recommendations, label: 'recommendations' },
              { icon: Star,     value: profile.total_reviews,          label: 'reviews'         },
              { icon: ThumbsUp, value: profile.total_votes_received,   label: 'votes received'  },
            ].map(({ icon: Icon, value, label }) => (
              <div key={label} className="flex items-center gap-1.5 text-sm text-muted-foreground">
                <Icon size={13} className="text-primary/60" />
                <span><strong className="text-foreground font-black">{value ?? 0}</strong> {label}</span>
              </div>
            ))}
            {!isGhost && (
              <div className="flex items-center gap-1.5 text-sm text-muted-foreground">
                <CalendarDays size={13} className="text-primary/60" />
                <span>Joined {formatDate(profile.member_since)}</span>
              </div>
            )}
          </div>

          {favoriteGenres.length > 0 && !isGhost && (
            <div className="flex flex-wrap gap-1.5">
              {favoriteGenres.map((genre: string) => (
                <Badge key={genre} variant="secondary" className="text-xs rounded-full">
                  {genre}
                </Badge>
              ))}
            </div>
          )}
        </div>
      </div>

      {/* ── Pinned Recommendation ── */}
      {pinnedRec && (
        <section className="space-y-3">
          <div className="flex items-center gap-2">
            <Pin size={14} className="text-amber-500" />
            <h2 className="text-sm font-black text-amber-600 uppercase tracking-wider">Pinned</h2>
          </div>
          <div className={`rounded-2xl border ${showGold ? 'border-amber-500/30 bg-amber-500/5' : 'border-border'}`}>
            <ComicCard recommendation={pinnedRec} />
          </div>
        </section>
      )}

      {/* ── Recommendations ── */}
      {recommendations.length > 0 ? (
        <section className="space-y-4">
          <div className="flex items-center gap-2">
            <h2 className="text-lg font-black">
              {isGhost ? 'Archived Recommendations' : 'Recent Recommendations'}
            </h2>
            <div className={`h-0.5 flex-1 bg-gradient-to-r ${
              isGhost  ? 'from-indigo-500/40' :
              showGold ? 'from-amber-500/40'  :
                         'from-primary/40'
            } to-transparent`} />
            {hasMore && (
              <Link
                href={`/profile/${profile.username}/recommendations`}
                className="flex items-center gap-1 text-xs font-bold text-primary hover:underline shrink-0"
              >
                See all {totalRecs}
                <ArrowRight size={12} />
              </Link>
            )}
          </div>
          <div className="grid grid-cols-2 sm:grid-cols-3 gap-4">
            {recommendations.map((rec) => (
              <ComicCard key={rec.id} recommendation={rec} />
            ))}
          </div>
          {hasMore && (
            <Link
              href={`/profile/${profile.username}/recommendations`}
              className="flex items-center justify-center gap-2 w-full py-3 rounded-2xl border border-border hover:border-primary/40 hover:bg-primary/5 text-sm font-semibold text-muted-foreground hover:text-primary transition-all duration-200"
            >
              See all {totalRecs} recommendations
              <ArrowRight size={14} />
            </Link>
          )}
        </section>
      ) : (
        <div className="relative flex flex-col items-center justify-center py-16 rounded-2xl border border-dashed border-border overflow-hidden">
          <div className="absolute -top-32 -right-32 w-96 h-96 rounded-full bg-primary/5 blur-3xl pointer-events-none" />
          <div className="absolute -bottom-32 -left-32 w-96 h-96 rounded-full bg-violet-500/5 blur-3xl pointer-events-none" />
          <div className="relative mb-6 w-36 h-36 mx-auto">
            <div className="absolute top-0 left-0 w-20 h-28 rounded-2xl border-2 border-border bg-card rotate-[-12deg] opacity-40" />
            <div className="absolute top-2 left-6 w-20 h-28 rounded-2xl border-2 border-border bg-card rotate-[-5deg] opacity-60" />
            <div className="absolute top-4 left-10 w-20 h-28 rounded-2xl border-2 border-primary/30 bg-card shadow-xl flex flex-col items-center justify-center gap-1">
              <BookOpen size={18} className="text-primary opacity-60" />
              <span className="text-[8px] font-bold text-muted-foreground uppercase tracking-widest text-center px-1">
                {isGhost ? 'Archive' : 'Empty'}
              </span>
            </div>
          </div>
          <p className="font-black text-sm">
            {isGhost ? 'The archive is empty' : 'No recommendations yet'}
          </p>
          <p className="text-xs text-muted-foreground mt-1 text-center px-4">
            {isGhost
              ? 'Comics find their way here when their recommenders move on. More always arrive.'
              : isOwner
              ? 'Share a comic you love and help the community find their next read.'
              : `@${profile.username} hasn't shared any recommendations yet.`
            }
          </p>
          {isOwner && !isGhost && (
            <Link href="/dashboard/recommendations/new" className="mt-5">
              <Button size="sm" className="gap-2 bg-gradient-to-r from-primary to-violet-500 border-0 hover:opacity-90 shadow-md shadow-primary/20">
                <PenLine size={13} />
                Submit your first rec
              </Button>
            </Link>
          )}
        </div>
      )}

      {/* ── Public Reading Lists — hidden for ghost ── */}
      {!isGhost && lists.length > 0 && (
        <section className="space-y-4">
          <div className="flex items-center gap-2">
            <h2 className="text-lg font-black">Reading Lists</h2>
            <div className="h-0.5 flex-1 bg-gradient-to-r from-teal-500/40 to-transparent" />
            <span className="text-xs text-muted-foreground shrink-0">{lists.length} public</span>
          </div>
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
            {lists.map((list) => (
              <Link
                key={list.id}
                href={`/lists/${list.id}`}
                className="group relative p-4 rounded-2xl border border-border bg-card hover:border-teal-500/40 hover:shadow-md transition-all duration-200 space-y-3"
              >
                <div className="flex items-start justify-between gap-3">
                  <div className="flex items-center gap-2.5 min-w-0">
                    <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-teal-500/20 to-cyan-500/10 border border-teal-500/20 flex items-center justify-center shrink-0">
                      <BookMarked size={14} className="text-teal-600" />
                    </div>
                    <div className="min-w-0">
                      <p className="text-sm font-black truncate group-hover:text-teal-600 transition-colors">
                        {list.name}
                      </p>
                      {list.description && (
                        <p className="text-xs text-muted-foreground line-clamp-1 mt-0.5">
                          {list.description}
                        </p>
                      )}
                    </div>
                  </div>
                  <Globe size={12} className="text-emerald-500 shrink-0 mt-1" />
                </div>
                <div className="flex items-center justify-between text-xs text-muted-foreground">
                  <div className="flex items-center gap-1">
                    <BookOpen size={11} className="text-teal-500/60" />
                    <span>
                      <strong className="text-foreground font-black">{list.item_count}</strong>{' '}
                      comic{list.item_count !== 1 ? 's' : ''}
                    </span>
                  </div>
                  <span className="text-[10px] font-bold text-teal-600 group-hover:translate-x-0.5 transition-transform">
                    View →
                  </span>
                </div>
              </Link>
            ))}
          </div>
        </section>
      )}
    </div>
  )
}
