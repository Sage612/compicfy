async function getInitialGrid(params: SearchParams) {
  const supabase = await createClient()

  const { data } = await (supabase as any).rpc('search_recommendations', {
    p_query:  params.q      || null,
    p_type:   params.type   || null,
    p_genre:  params.genre  || null,
    p_status: params.status || null,
    p_sort:   params.sort   ?? 'score',
    p_adult:  false,
    p_limit:  PAGE_SIZE,
    p_offset: 0,
  })

  const { count } = await supabase
    .from('recommendations')
    .select('*', { count: 'exact', head: true })
    .eq('is_approved', true)
    .not('content_rating', 'in', '("adult","mature")')

  return {
    items: (data as Recommendation[]) ?? [],
    total: count ?? 0,
  }
}

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { id } = await params
  const supabase = await createClient()
  const { data } = await supabase
    .from('recommendations')
    .select('title, description, type, genres, author, cover_url, status')
    .eq('id', id)
    .single()

  const rec = data as Pick<
    Recommendation,
    'title' | 'description' | 'type' | 'genres' | 'author' | 'cover_url' | 'status'
  > | null
  if (!rec) return { title: 'Comic | Compicfy' }

  const typeLabel = rec.type
    ? rec.type.charAt(0).toUpperCase() + rec.type.slice(1)
    : 'Comic'

  const genreList = rec.genres?.slice(0, 3).join(', ') ?? ''

  // Title optimised for search — e.g. "Solo Leveling Manhwa | Compicfy"
  const title = `${rec.title} ${typeLabel} | Compicfy`

  // Description targets "where to read", "review" search intent
  const descParts = [
    `${rec.title} is a ${typeLabel.toLowerCase()}`,
    rec.author ? `by ${rec.author}` : null,
    genreList ? `in the genres ${genreList}` : null,
    rec.status ? `— currently ${rec.status}` : null,
    rec.description
      ? `. ${rec.description.slice(0, 100)}`
      : '. Discover where to read it and what the community thinks.',
  ].filter(Boolean).join(' ')

  const description = descParts.slice(0, 160)

  // Keywords target high-intent searches
  const keywords = [
    rec.title,
    `${rec.title} ${rec.type}`,
    `${rec.title} review`,
    `${rec.title} where to read`,
    `${rec.title} recommendation`,
    ...(rec.genres ?? []),
    rec.author ?? '',
    typeLabel,
    'manga recommendation',
    'manhwa recommendation',
    'webtoon recommendation',
  ].filter(Boolean)

  return {
    title,
    description,
    keywords,
    openGraph: {
      title,
      description,
      type: 'article',
      url: `https://compicfy.cloud/comic/${id}`,
      images: rec.cover_url
        ? [
            {
              url: rec.cover_url,
              width: 400,
              height: 600,
              alt: `${rec.title} cover`,
            },
          ]
        : [],
    },
    twitter: {
      card: 'summary_large_image',
      title,
      description,
      images: rec.cover_url ? [rec.cover_url] : [],
    },
    alternates: {
      canonical: `https://compicfy.cloud/comic/${id}`,
    },
  }
}

interface Props {
  searchParams: Promise<{
    category?: string
    q?: string
    page?: string
  }>
}

async function NewsGrid({ searchParams }: Props) {
  const params = await searchParams
  const supabase = await createClient()

  const page = parseInt(params.page ?? '1')
  const limit = 12
  const offset = (page - 1) * limit

  let query = supabase
    .from('news')
    .select('*', { count: 'exact' })
    .eq('is_published', true)
    .order('published_at', { ascending: false })

  if (params.category) query = query.eq('category', params.category)
  if (params.q)        query = query.ilike('title', `%${params.q}%`)

  query = query.range(offset, offset + limit - 1)

  const { data, count } = await query
  const news = (data as News[]) ?? []
  const totalPages = Math.ceil((count ?? 0) / limit)

  if (!news.length) {
    return (
      <div className="text-center py-20">
        <div className="w-16 h-16 rounded-2xl bg-muted flex items-center justify-center mx-auto mb-4">
          <Newspaper size={28} className="text-muted-foreground opacity-40" />
        </div>
        <p className="font-black text-lg">No news found</p>
        <p className="text-sm text-muted-foreground mt-1">Check back soon for updates!</p>
      </div>
    )
  }

  return (
    <div className="space-y-6">
      <p className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">
        {count} article{count !== 1 ? 's' : ''}
      </p>
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
        {news.map((item) => (
          <NewsCard key={item.id} news={item} />
        ))}
      </div>

      {totalPages > 1 && (
        <div className="flex items-center justify-center gap-3 pt-4">
          {page > 1 && (
            <a
              href={`/news?page=${page - 1}`}
              className="px-4 py-2 rounded-xl border border-border bg-card text-sm font-semibold hover:border-primary/50 transition-colors"
            >
              ← Previous
            </a>
          )}
          <span className="text-sm text-muted-foreground font-medium">Page {page} of {totalPages}</span>
          {page < totalPages && (
            <a
              href={`/news?page=${page + 1}`}
              className="px-4 py-2 rounded-xl border border-border bg-card text-sm font-semibold hover:border-primary/50 transition-colors"
            >
              Next →
            </a>
          )}
        </div>
      )}
    </div>
  )
}
