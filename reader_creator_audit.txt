    .from('follows').select('following_id').eq('follower_id', user.id)

  const followingIds = (followData ?? []).map((f: any) => f.following_id)

  const [
    { count: listsCount },
    { count: savedCount },

    supabase.from('reading_lists').select('*', { count: 'exact', head: true }).eq('user_id', user.id),
    supabase.from('saves').select('*', { count: 'exact', head: true }).eq('user_id', user.id),
    supabase.from('follows').select('*', { count: 'exact', head: true }).eq('follower_id', user.id),
    supabase.from('reviews').select('*', { count: 'exact', head: true }).eq('user_id', user.id),

      .select('id, list_id, status, progress_chapter, last_read_at, added_at, reading_lists!inner(user_id), recommendations(id, title, cover_url, type, status, chapter_count)')

      .order('added_at', { ascending: false }).limit(6),
    followingIds.length > 0

          .from('recommendations')
          .select(`id, title, cover_url, type, created_at, user_id, profiles!recommendations_user_id_fkey(username, display_name, avatar_url)`)

          .eq('is_approved', true)
          .not('content_rating', 'in', '("adult","mature")')
          .order('created_at', { ascending: false })

  const feedItems = (rawFeedItems ?? []).map((r: any) => {

    { label: 'Reviews Written', value: reviewCount ?? 0,                   icon: Star,       href: '/reader/dashboard/reviews',         bg: 'bg-amber-500/10',  text: 'text-amber-500' },

    { href: '/reader/dashboard/lists',               icon: BookMarked, label: 'Reading Lists',            desc: 'Manage your reading collection',  bg: 'bg-teal-500/10',   text: 'text-teal-500'   },
    { href: '/browse',                               icon: Flame,      label: 'Browse Comics',            desc: 'Discover new recommendations',    bg: 'bg-rose-500/10',   text: 'text-rose-500'   },

import { createClient } from '@/lib/supabase/server'
import { redirect } from 'next/navigation'
import { Header } from '@/components/layout/Header'
import { AdminFloatingButton } from '@/components/admin/AdminFloatingButton'
import { IncompleteRecNudge } from '@/components/dashboard/IncompleteRecNudge'
import { UpgradeBanner } from '@/components/pro/UpgradeBanner'
import { DashboardNav } from '@/components/layout/DashboardNav'
import { DashboardMobileNav } from '@/components/layout/DashboardMobileNav'

export default async function ReaderDashboardLayout({
  children,
}: {
  children: React.ReactNode
}) {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) redirect('/login')

  const { data: profile } = await supabase
    .from('profiles')
    .select('role, is_pro, pro_expires_at, recommendation_count_this_month, user_type')
    .eq('id', user.id)
    .single() as { data: any }

  if (!profile) redirect('/login')

  if (profile.user_type === 'creator') redirect('/creator/dashboard')

  const isAdmin      = ['admin', 'moderator'].includes(profile?.role ?? '')
  const isPro        = profile?.is_pro &&
    (!profile.pro_expires_at || new Date(profile.pro_expires_at) > new Date())
  const isPrivileged = isAdmin || isPro
  const recCount     = profile?.recommendation_count_this_month ?? 0

  if (isPro && !isAdmin) redirect('/pro/reader/dashboard')

  return (
    <div className="min-h-screen flex flex-col bg-background">
      <Header />
      <div className="flex flex-1 pt-20">
        <aside className="dashboard-sidebar hidden md:flex flex-col w-56 shrink-0 border-r border-border p-3 gap-1">
          <DashboardNav />
        </aside>

        <main className="dashboard-main flex-1 min-w-0 overflow-auto pb-24 md:pb-0">
          {!isPrivileged && recCount >= 20 && (
            <div className="px-4 pt-4">
              <UpgradeBanner
                current={recCount}
                limit={50}
                feature="recommendations this month"
                message="You're getting close to your monthly limit"
                dismissible
              />
            </div>
          )}
          <div className="px-4 pt-4">
            <IncompleteRecNudge />
          </div>
          {children}
        </main>
      </div>

      <DashboardMobileNav />
      {isAdmin && <AdminFloatingButton />}
    </div>
  )
}

                {f}
              </div>
            ))}
          </div>
          <Link href="/pricing">
            <Button className="w-full gap-2 bg-gradient-to-r from-amber-400 to-orange-500 border-0 text-white shadow-md shadow-amber-500/20">
              <Crown size={14} />
              Upgrade to Pro
              <ArrowRight size={13} />
            </Button>
          </Link>
        </div>
      )}

      {isPro && !isAdmin && <BillingClient profile={profile} subscription={subscription} email={user.email ?? ''} />}
    </div>
  )
}
