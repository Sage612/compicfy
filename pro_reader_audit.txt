export default async function ProReaderDashboardPage() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) redirect('/login')

  const { data: profileData } = await supabase
    .from('profiles').select('*').eq('id', user.id).single()

  if (!profileData) redirect('/login')
  const profile = profileData as any

  const isAdmin = profile.role === 'admin'
  const isPro   = isAdmin ||
    (profile.is_pro && (!profile.pro_expires_at || new Date(profile.pro_expires_at) > new Date()))

  if (!isPro) redirect('/pricing?reason=pro_required')

  const { data: followData } = await supabase
    .from('follows').select('following_id').eq('follower_id', user.id)
  const followingIds = (followData ?? []).map((f: any) => f.following_id)

  const [
    { count: totalRecs },
    { count: savedCount },
    { count: listsCount },
    { count: followingCount },
    { count: reviewCount },
    { data: topRecs },
    { data: leaderboardData },
    { count: viewsThisMonth },
    { data: currentlyReading },
    { data: rawFeedItems },
    { data: listsForTracker },
  ] = await Promise.all([
    supabase.from('recommendations').select('*', { count: 'exact', head: true }).eq('user_id', user.id).eq('is_approved', true),
    supabase.from('saves').select('*', { count: 'exact', head: true }).eq('user_id', user.id),
    supabase.from('reading_lists').select('*', { count: 'exact', head: true }).eq('user_id', user.id),
    supabase.from('follows').select('*', { count: 'exact', head: true }).eq('follower_id', user.id),
    supabase.from('reviews').select('*', { count: 'exact', head: true }).eq('user_id', user.id),
    supabase.from('recommendations').select('id, title, cover_url, upvotes, save_count, view_count, type').eq('user_id', user.id).eq('is_approved', true).order('score', { ascending: false }).limit(3),
    (supabase as any).from('pro_leaderboard').select('rank, score').eq('user_id', user.id).order('month', { ascending: false }).limit(1).maybeSingle(),
    (supabase as any).from('profile_views').select('*', { count: 'exact', head: true }).eq('profile_id', user.id).gte('viewed_at', new Date(new Date().getFullYear(), new Date().getMonth(), 1).toISOString()),
    supabase.from('reading_list_items')
      .select('id, list_id, status, progress_chapter, last_read_at, added_at, reading_lists!inner(user_id), recommendations(id, title, cover_url, type, status, chapter_count)')
      .eq('reading_lists.user_id', user.id).eq('status', 'reading')
      .order('added_at', { ascending: false }).limit(6),
    followingIds.length > 0
      ? supabase.from('recommendations')
          .select('id, title, cover_url, type, created_at, user_id, profiles!recommendations_user_id_fkey(username, display_name, avatar_url)')
          .in('user_id', followingIds).eq('is_approved', true)
          .not('content_rating', 'in', '("adult","mature")')
          .order('created_at', { ascending: false }).limit(4)
      : Promise.resolve({ data: [] }),
    supabase.from('reading_lists').select('id, name').eq('user_id', user.id).order('created_at', { ascending: false }),
  ])

  const feedItems = (rawFeedItems ?? []).map((r: any) => {
    const p = Array.isArray(r.profiles) ? r.profiles[0] : r.profiles
    return { ...r, profiles: p ?? null }
  })

  const leaderboardEntry  = leaderboardData as { rank: number; score: number } | null
  const initials          = (profile.display_name || profile.username || 'U').slice(0, 2).toUpperCase()
  const isFollowingNobody = (followingCount ?? 0) === 0
  const hasReadingItems   = (currentlyReading ?? []).length > 0

  const proTools = [
    { href: '/pro/reader/dashboard/analytics',   icon: BarChart2, label: 'Analytics',     desc: 'See your rec performance',           color: 'bg-violet-500/10', text: 'text-violet-500' },
    { href: '/pro/reader/dashboard/analytics',   icon: Eye,       label: 'Profile Views', desc: `${viewsThisMonth ?? 0} this month`,  color: 'bg-blue-500/10',   text: 'text-blue-500'   },
    { href: '/pro/reader/dashboard/export',      icon: Download,  label: 'Export Lists',  desc: 'Download as CSV or PDF',             color: 'bg-primary/10',    text: 'text-primary'    },
    { href: '/pro/reader/dashboard/leaderboard', icon: Trophy,    label: 'Leaderboard',   desc: leaderboardEntry?.rank ? `Rank #${leaderboardEntry.rank}` : 'See your rank', color: 'bg-amber-500/10', text: 'text-amber-500' },
  ]

  const statsRow = [
    { label: 'Recommendations', value: totalRecs ?? 0,      icon: BookOpen,   color: 'text-violet-500', bg: 'bg-violet-500/10', href: '/pro/reader/dashboard/recommendations' },
    { label: 'Saved Comics',    value: savedCount ?? 0,     icon: Bookmark,   color: 'text-primary',    bg: 'bg-primary/10',    href: '/pro/reader/dashboard/saved'           },
    { label: 'Reading Lists',   value: listsCount ?? 0,     icon: BookMarked, color: 'text-teal-500',   bg: 'bg-teal-500/10',   href: '/pro/reader/dashboard/lists'           },
    { label: 'Reviews Written', value: reviewCount ?? 0,    icon: Star,       color: 'text-amber-500',  bg: 'bg-amber-500/10',  href: '/pro/reader/dashboard/reviews'         },
  ]

  return (
    <div className="max-w-4xl mx-auto px-4 py-8 space-y-8">

      {/* Gold welcome banner */}
      <div className="relative overflow-hidden rounded-2xl border border-amber-500/30 bg-gradient-to-br from-amber-500/10 via-orange-500/5 to-transparent p-6 md:p-8">
        <div className="absolute -top-12 -right-12 w-48 h-48 rounded-full bg-amber-500/10 blur-2xl pointer-events-none" />
        <div className="absolute top-0 left-0 right-0 h-px bg-gradient-to-r from-transparent via-amber-500/40 to-transparent" />
        <div className="relative flex flex-col sm:flex-row items-start sm:items-center gap-5">
          <ProAvatar src={profile.avatar_url} fallback={initials} size="lg" isPro={isPro} isAdmin={isAdmin} />
          <div className="flex-1 min-w-0 space-y-1">
            <div className="flex flex-wrap items-center gap-2">
              <h1 className="text-xl font-black truncate">Hey, {profile.display_name || profile.username}!</h1>
              <ProBadge role={profile.role} isPro={isPro} />
              {profile.verified_recommender && <VerifiedBadge />}
            </div>
            <p className="text-sm text-muted-foreground">
              {isAdmin ? 'Admin · Lifetime Pro access' : <>Pro Reader · {profile.pro_expires_at ? `Renews ${formatDate(profile.pro_expires_at)}` : 'Lifetime access'}</>}
            </p>
            {leaderboardEntry?.rank && (
              <p className="text-xs text-amber-600 font-semibold flex items-center gap-1">
                <Trophy size={11} />Leaderboard Rank #{leaderboardEntry.rank} this month
              </p>
            )}
          </div>
          <div className="flex gap-2 shrink-0">
            <Link href="/pro/reader/dashboard/settings">
              <Button variant="outline" size="sm" className="gap-1.5 border-amber-500/30 hover:border-amber-500/60">
                <Settings size={13} />Settings
              </Button>
            </Link>
            <Link href="/pro/reader/dashboard/recommendations/new">
              <Button size="sm" className="gap-1.5 bg-gradient-to-r from-amber-400 to-orange-500 border-0 text-white shadow-md shadow-amber-500/20">
                <BookOpen size={13} />Recommend
              </Button>
            </Link>
          </div>
        </div>
      </div>

      <ProfileCompletionBanner hasAvatar={!!profile.avatar_url} hasBio={!!profile.bio} hasWebsite={!!profile.website} />

      {/* Stats */}
      <div>
        <h2 className="text-sm font-semibold text-muted-foreground uppercase tracking-wider mb-3">Your Stats</h2>

import { createClient } from '@/lib/supabase/server'
import { redirect } from 'next/navigation'
import { ProAnalyticsClient } from '@/components/pro/ProAnalyticsClient'
import type { Metadata } from 'next'

export const metadata: Metadata = { title: 'Analytics | Pro' }

export default async function ProCreatorAnalyticsPage() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) redirect('/login')

  const [
    { data: recommendations },
    { data: analytics },
  ] = await Promise.all([
    supabase
      .from('recommendations')
      .select('id, title, cover_url, type, upvotes, downvotes, save_count, view_count, review_count, score, created_at')
      .eq('user_id', user.id)
      .eq('is_approved', true)
      .order('score', { ascending: false }),
    (supabase as any)
      .from('recommendation_analytics')
      .select('recommendation_id, date, views, saves, upvotes, downvotes')
      .eq('user_id', user.id)
      .order('date', { ascending: false })
      .limit(500),
  ])

  return (
    <div className="max-w-4xl mx-auto px-4 py-8">
      <ProAnalyticsClient
        recommendations={recommendations ?? []}
        analytics={analytics ?? []}
      />
    </div>
  )
}

import { useEffect, useState } from 'react'

import { PenLine, X, ChevronRight } from 'lucide-react'
import { createClient } from '@/lib/supabase/client'
import { cn } from '@/lib/utils'

interface DraftRec {
  id:    string
  title: string
}

const SESSION_KEY = 'compicfy_nudge_dismissed'

export function IncompleteRecNudge() {
  const [drafts,  setDrafts]  = useState<DraftRec[]>([])
  const [visible, setVisible] = useState(false)
  const [loading, setLoading] = useState(true)
  const [isPro,   setIsPro]   = useState(false)

  useEffect(() => {
    try {
      if (typeof window !== 'undefined' && sessionStorage.getItem(SESSION_KEY)) {
        setLoading(false)
        return
      }
    } catch {}

    const run = async () => {
      try {
        // Check pro status so we can build the correct edit link
        const supabase = createClient()
        const { data: { user } } = await supabase.auth.getUser()

        if (user) {
          const { data: profile } = await (supabase as any)
            .from('profiles')
            .select('is_pro, pro_expires_at, role')
            .eq('id', user.id)
            .single()

          if (profile) {
            setIsPro(
              profile.role === 'admin' ||
              (profile.is_pro &&
                (!profile.pro_expires_at || new Date(profile.pro_expires_at) > new Date()))
            )
          }
        }

        const res = await fetch('/api/recommendations/drafts', { cache: 'no-store' })
        if (!res.ok) return

        const data = await res.json()
        const list: DraftRec[] = data.drafts ?? []

        if (list.length > 0) {
          setDrafts(list)
          setVisible(true)
        }
      } catch (err) {
        console.warn('[IncompleteRecNudge] failed:', err)
      } finally {
        setLoading(false)
      }
    }

    run()
  }, [])

  const dismiss = () => {
    try { sessionStorage.setItem(SESSION_KEY, '1') } catch {}
    setVisible(false)
  }

  // Edit route depends on whether user is pro or not
  const editHref = (id: string) =>
    isPro
      ? `/pro/reader/dashboard/recommendations/${id}/edit`
      : `/reader/dashboard/recommendations/${id}/edit`

  if (loading || !visible || drafts.length === 0) return null

  return (
    <div className={cn(
      'flex items-start gap-3 px-4 py-3 rounded-2xl border',
      'border-amber-500/25 bg-amber-500/5'
    )}>
      <div className="w-8 h-8 rounded-xl bg-gradient-to-br from-amber-500 to-orange-500 flex items-center justify-center shrink-0 shadow-sm mt-0.5">

        <p className="text-sm font-black text-amber-600 leading-tight">

            : `${drafts.length} recommendations are missing your take`}
        </p>
        <p className="text-xs text-muted-foreground leading-snug">
          The &quot;why do you recommend it?&quot; field is empty.

              href={editHref(draft.id)}

          </div>
          <div className="min-w-0">
            <h1 className="text-xl font-black tracking-tight leading-tight truncate">My Recommendations</h1>
            <div className="flex items-center gap-1.5 mt-0.5">
              <Crown size={11} className="text-amber-500 fill-amber-500" />
              <span className="text-xs text-amber-600 font-semibold">Unlimited · Pro</span>
            </div>
          </div>
        </div>
        <Link href="/pro/recommendations/new" className="shrink-0">
          <Button className="gap-2 bg-primary hover:bg-primary/90 border-0 shadow-md shadow-primary/20">
            <Plus size={15} />
            Add New
          </Button>
        </Link>
      </div>

      {/* Search + filter */}
      {recs.length > 0 && (
        <div className="space-y-3">
          <div className="relative">
            <Search size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground pointer-events-none" />
            <Input
              value={search}
              onChange={e => setSearch(e.target.value)}
              placeholder="Search your recommendations…"
              className="pl-9 h-10 bg-card"
            />
            {search && (
              <button onClick={() => setSearch('')} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground text-xs">✕</button>
            )}
          </div>
          <div className="flex gap-1.5 flex-wrap">
            {FILTER_TABS.map(({ key, label }) => (
              <button
                key={key}
                onClick={() => setFilter(key)}
                className={cn(
                  'inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl text-xs font-semibold transition-all border',
                  filter === key
                    ? 'bg-amber-500/10 text-amber-600 border-amber-500/30'
                    : 'bg-card text-muted-foreground border-border hover:border-amber-500/30 hover:text-foreground'
                )}
              >
                {label}
                <span className={cn(
                  'inline-flex items-center justify-center w-4 h-4 rounded-full text-[10px] font-black',
                  filter === key ? 'bg-amber-500 text-white' : 'bg-muted text-muted-foreground'
                )}>
                  {counts[key]}
                </span>
              </button>
            ))}
          </div>
        </div>
      )}

      {/* Empty — no recs */}
      {recs.length === 0 && (
        <div className="relative overflow-hidden text-center py-20 rounded-2xl border border-dashed border-border bg-card">
          <div className="absolute -top-10 -right-10 w-40 h-40 rounded-full bg-amber-500/5 blur-2xl pointer-events-none" />
          <div className="absolute -bottom-10 -left-10 w-40 h-40 rounded-full bg-violet-500/5 blur-2xl pointer-events-none" />
          <div className="relative w-28 h-28 mx-auto mb-6">
            <div className="absolute top-0 left-2 w-16 h-22 rounded-xl border-2 border-border bg-muted/60 rotate-[-8deg]" />
            <div className="absolute top-2 left-5 w-16 h-22 rounded-xl border-2 border-border bg-muted/80 rotate-[-3deg]" />
            <div className="absolute top-4 left-8 w-16 h-22 rounded-xl border-2 border-amber-500/30 bg-card flex items-center justify-center shadow-lg">
              <BookOpen size={20} className="text-amber-500/50" />
            </div>
          </div>
          <h3 className="font-black text-xl mb-2">No recommendations yet</h3>
          <p className="text-sm text-muted-foreground mb-6 max-w-xs mx-auto leading-relaxed">
            As a Pro member you have unlimited monthly submissions. Start sharing great comics.
          </p>
          <div className="flex flex-wrap justify-center gap-3">
            <Link href="/pro/recommendations/new">
              <Button className="gap-2 bg-primary hover:bg-primary/90 border-0 shadow-md shadow-primary/20">
                <BookOpen size={14} />Submit a Recommendation
              </Button>
            </Link>
            <Link href="/browse">
              <Button variant="outline" className="gap-2 hover:border-amber-500/40 hover:text-amber-600 transition-colors">
                <Radar size={14} />Browse Comics
              </Button>
            </Link>
          </div>
        </div>
      )}

      {/* No results for search/filter */}
      {recs.length > 0 && visible.length === 0 && (
        <div className="flex flex-col items-center justify-center py-16 rounded-2xl border border-dashed border-border bg-card text-center space-y-3">
          <div className="w-12 h-12 rounded-2xl bg-muted/40 flex items-center justify-center">
            <Search size={20} className="text-muted-foreground opacity-50" />
          </div>
          <h3 className="font-black text-sm">No matches found</h3>
          <p className="text-xs text-muted-foreground">Try a different search term or filter</p>
          <Button variant="ghost" size="sm" onClick={() => { setSearch(''); setFilter('all') }} className="text-xs">
            Clear filters
          </Button>
        </div>
      )}

      {/* Cards */}
      {visible.length > 0 && (
        <div className="space-y-3">
          {visible.map(rec => {
            const status       = getStatus(rec)
            const StatusIcon   = status.icon
            const typeStyle    = TYPE_COLORS[rec.type] ?? 'bg-muted text-muted-foreground border-border'
            const isDeleting   = deletingId === rec.id
            const isThisPinned = pinnedId === rec.id

            return (
              <div
                key={rec.id}
                className={cn(
                  'group flex gap-4 p-4 rounded-2xl border bg-card hover:shadow-md transition-all duration-200',
                  isThisPinned
                    ? 'border-amber-500/40 hover:border-amber-500/60'
                    : 'border-border hover:border-amber-500/30'
                )}
              >
                {/* Cover */}
                <div className="w-14 h-20 rounded-xl overflow-hidden bg-muted shrink-0 shadow-sm">
                  {rec.cover_url
                    ? <img src={rec.cover_url} alt={rec.title} className="w-full h-full object-cover" />
                    : <div className="w-full h-full flex items-center justify-center"><BookOpen size={16} className="text-muted-foreground opacity-40" /></div>
                  }
                </div>

                {/* Info */}
                <div className="flex-1 min-w-0 space-y-2">
                  <div className="flex items-start justify-between gap-2">
                    <div className="min-w-0 flex-1">
                      <h3 className="font-black truncate text-sm leading-snug group-hover:text-amber-600 transition-colors">
                        {rec.title}
                      </h3>
                      <div className="flex items-center gap-2 mt-1.5 flex-wrap">
                        <span className={`inline-flex items-center px-2 py-0.5 rounded-full text-[10px] font-bold border capitalize ${typeStyle}`}>
                          {rec.type}
                        </span>
                        <span className="text-xs text-muted-foreground">{formatDate(rec.created_at)}</span>
                        {rec.updated_at !== rec.created_at && (
                          <span className="text-[10px] text-muted-foreground/60 italic">edited</span>
                        )}
                        {isThisPinned && (
                          <span className="text-[10px] text-amber-600 font-bold">📌 Pinned</span>
                        )}
                      </div>
                    </div>
                    <span className={`inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-xs font-semibold border shrink-0 ${status.bg} ${status.color}`}>
                      <StatusIcon size={11} />
                      <span className="hidden xs:inline">{status.label}</span>
                    </span>
                  </div>

                  {/* Stats */}
                  <div className="flex items-center gap-3 text-xs text-muted-foreground flex-wrap">
                    <span className="flex items-center gap-1"><ThumbsUp      size={11} className="text-emerald-500" />{rec.upvotes}</span>
                    <span className="flex items-center gap-1"><ThumbsDown    size={11} className="text-red-400"     />{rec.downvotes}</span>
                    <span className="flex items-center gap-1"><MessageSquare size={11} className="text-blue-400"    />{rec.review_count}</span>
                    <span className="flex items-center gap-1"><Bookmark      size={11} className="text-violet-400"  />{rec.save_count}</span>
                  </div>

                  {/* Actions */}
                  <div className="flex items-center gap-2 pt-0.5 flex-wrap md:opacity-0 md:group-hover:opacity-100 transition-opacity">
                    <Link href={`/pro/recommendations/${rec.id}/edit`}>
                      <Button variant="outline" size="sm" className="h-7 text-xs gap-1.5 hover:border-amber-500/40 hover:text-amber-600">
                        <Pencil size={11} />Edit
                      </Button>
                    </Link>
                    <Button
                      variant="outline" size="sm"
                      onClick={() => handleDelete(rec.id, rec.title)}
                      disabled={isDeleting}
                      className="h-7 text-xs gap-1.5 hover:border-destructive/40 hover:text-destructive"
                    >
                      {isDeleting ? <Loader2 size={11} className="animate-spin" /> : <Trash2 size={11} />}
                      {isDeleting ? 'Deleting…' : 'Delete'}
                    </Button>
                    {rec.is_approved && (
                      <>
                        <Link href={`/comic/${rec.id}`}>
                          <Button variant="ghost" size="sm" className="h-7 text-xs gap-1.5 text-muted-foreground hover:text-foreground">
                            View →
                          </Button>
                        </Link>
                        {/* Pin button — approved recs only */}
                        <PinButton
                          recommendationId={rec.id}
                          isPinned={isThisPinned}
                          size="xs"
                          onPinChange={(newPinnedId) => setPinnedId(newPinnedId)}
                        />
                      </>
                    )}
                  </div>

                  {/* Rejection + appeal */}
                  {rec.rejection_reason && (
                    <div className="space-y-1.5 pt-1">
                      <p className="text-xs text-red-500 bg-red-500/5 border border-red-500/10 rounded-lg px-2.5 py-1.5 leading-relaxed">
                        Reason: {rec.rejection_reason}
                      </p>
                      {(!rec.appeal_status || rec.appeal_status === 'none') && <AppealButton recommendationId={rec.id} />}
                      {rec.appeal_status === 'pending'  && <span className="text-xs text-amber-500">🚩 Appeal pending review</span>}
                      {rec.appeal_status === 'approved' && <span className="text-xs text-emerald-500">✅ Appeal approved</span>}
                      {rec.appeal_status === 'rejected' && <span className="text-xs text-muted-foreground">Appeal rejected</span>}
                    </div>
                  )}
                </div>
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}
